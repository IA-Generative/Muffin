{{- end }}
{{- if $component.ingress.enabled }}

    kubectl get ingress -n {{ $.Release.Namespace }} -l app.kubernetes.io/instance={{ $.Release.Name }},app.kubernetes.io/component={{ $componentName }}
{{- end }}
{{- if $component.metrics.enabled }}

  Metrics enabled (ServiceMonitor):
    kubectl get servicemonitor -n {{ $.Release.Namespace }} -l app.kubernetes.io/instance={{ $.Release.Name }},app.kubernetes.io/component={{ $componentName }}

  kubectl get all -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}
